        function evaluateCurrentSlot() {
            if (document.getElementById('anyTimeToggle').checked) {
                return { allowed: true, slotType: "ANYTIME" };
            }

            const now = new Date();
            const currentMins = now.getHours() * 60 + now.getMinutes();

            const parseMins = (timeStr) => {
                if (!timeStr) return 0;
                const parts = timeStr.split(':');
                return parseInt(parts[0], 10) * 60 + parseInt(parts[1], 10);
            };

            let slotType = null;
            let activeRangesStr = [];

            if (document.getElementById('slot1Enable').checked) {
                const start = parseMins(document.getElementById('slot1Start').value);
                const end = parseMins(document.getElementById('slot1End').value);
                activeRangesStr.push(`Morning (${document.getElementById('slot1Start').value} to ${document.getElementById('slot1End').value})`);
                if (currentMins >= start && currentMins <= end) slotType = "MORNING";
            }

            if (document.getElementById('slot2Enable').checked) {
                const start = parseMins(document.getElementById('slot2Start').value);
                const end = parseMins(document.getElementById('slot2End').value);
                activeRangesStr.push(`Evening (${document.getElementById('slot2Start').value} to ${document.getElementById('slot2End').value})`);
                if (currentMins >= start && currentMins <= end) slotType = "EVENING";
            }

            if (slotType) {
                return { allowed: true, slotType: slotType };
            } else {
                return { allowed: false, ranges: activeRangesStr.join(" OR ") };
            }
        }

        function detectFaceBox(imageData, width, height) {
            const data = imageData.data;
            let minX = width, minY = height, maxX = 0, maxY = 0;
            let skinCount = 0;

            for (let y = 0; y < height; y += 2) {
                for (let x = 0; x < width; x += 2) {
                    const idx = (y * width + x) * 4;
                    const r = data[idx], g = data[idx + 1], b = data[idx + 2];

                    if (r > 80 && g > 35 && b > 15 && (Math.max(r, g, b) - Math.min(r, g, b) > 15) && Math.abs(r - g) > 15 && r > g && r > b) {
                        skinCount++;
                        if (x < minX) minX = x;
                        if (x > maxX) maxX = x;
                        if (y < minY) minY = y;
                        if (y > maxY) maxY = y;
                    }
                }
            }

            const faceWidth = maxX - minX;
            const faceHeight = maxY - minY;

            if (skinCount < 180 || faceWidth < 28 || faceHeight < 28 || faceWidth / faceHeight > 1.8) {
                return null;
            }

            const padX = Math.floor(faceWidth * 0.12);
            const padY = Math.floor(faceHeight * 0.15);

            return {
                x: Math.max(0, minX - padX),
                y: Math.max(0, minY - padY),
                w: Math.min(width - minX, faceWidth + padX * 2),
                h: Math.min(height - minY, faceHeight + padY * 2)
            };
        }

        function extractBiometricDescriptor(ctx, box) {
            const size = 64;
            const sampleCanvas = document.createElement('canvas');
            sampleCanvas.width = size;
            sampleCanvas.height = size;
            const sampleCtx = sampleCanvas.getContext('2d');

            sampleCtx.drawImage(ctx.canvas, box.x, box.y, box.w, box.h, 0, 0, size, size);
            const imgData = sampleCtx.getImageData(0, 0, size, size).data;

            const vector = [];
            let mean = 0;

            for (let i = 0; i < imgData.length; i += 4) {
                const gray = 0.299 * imgData[i] + 0.587 * imgData[i + 1] + 0.114 * imgData[i + 2];
                vector.push(gray);
                mean += gray;
            }
            mean /= vector.length;

            let variance = 0;
            for (let i = 0; i < vector.length; i++) {
                vector[i] -= mean;
                variance += vector[i] * vector[i];
            }
            const stdDev = Math.sqrt(variance) || 1;
            for (let i = 0; i < vector.length; i++) {
                vector[i] /= stdDev;
            }

            return vector;
        }

        function matchDescriptors(vecA, vecB) {
            if (!vecA || !vecB || vecA.length !== vecB.length) return 0;
            let distSq = 0;
            for (let i = 0; i < vecA.length; i++) {
                const diff = vecA[i] - vecB[i];
                distSq += diff * diff;
            }
            const dist = Math.sqrt(distSq);
            
            const rmsDist = dist / Math.sqrt(vecA.length);
            const conf = Math.round((1 - (rmsDist / 1.0)) * 100);
            return Math.max(0, Math.min(100, conf));
        }

        let lastAttTime = performance.now();
        let attFrameCounter = 0;

        function processAttendanceFrame() {
            if (!isAttCamRunning || activePanel !== 'attendancePanel') return;

            const now = performance.now();
            attFrameCounter++;
            if (now - lastAttTime >= 1000) {
                document.getElementById('attFps').innerText = `FPS: ${attFrameCounter}`;
                attFrameCounter = 0;
                lastAttTime = now;
            }

            workCtx.drawImage(attVideo, 0, 0, workCanvas.width, workCanvas.height);
            const frameData = workCtx.getImageData(0, 0, workCanvas.width, workCanvas.height);
            const rawBox = detectFaceBox(frameData, workCanvas.width, workCanvas.height);

            attCtx.clearRect(0, 0, attOverlay.width, attOverlay.height);

            if (rawBox && isScannerActive) {
                const scaleX = attOverlay.width / workCanvas.width;
                const scaleY = attOverlay.height / workCanvas.height;

                const box = {
                    x: rawBox.x * scaleX,
                    y: rawBox.y * scaleY,
                    w: rawBox.w * scaleX,
                    h: rawBox.h * scaleY
                };

                const liveDescriptor = extractBiometricDescriptor(workCtx, rawBox);

                let bestMatch = null;
                let maxConf = 0;

                enrolledStudents.forEach(student => {
                    const conf = matchDescriptors(liveDescriptor, student.descriptor);
                    if (conf > maxConf) {
                        maxConf = conf;
                        bestMatch = student;
                    }
                });

                if (bestMatch && maxConf >= 95) {
                    resetUnauthorizedState();
                    drawBox(attCtx, box, `${bestMatch.name} (${maxConf}%)`, '#10b981');
                    
                    document.getElementById('liveMatchAlert').style.display = 'block';
                    document.getElementById('matchName').innerText = bestMatch.name;
                    document.getElementById('matchMeta').innerText = `Roll: ${bestMatch.roll} | Match Confidence: ${maxConf}%`;

                    currentlyRecognizedStudent = bestMatch;
                    currentlyRecognizedConf = maxConf;
                } else {
                    currentlyRecognizedStudent = null;
                    currentlyRecognizedConf = 0;

                    document.getElementById('liveMatchAlert').style.display = 'none';
                    drawBox(attCtx, box, `UNAUTHORIZED (<95% Match)`, "#ef4444");

                    unauthorizedState.facePresent = true;
                    triggerThreeBlinksAlert();
                }
            } else {
                currentlyRecognizedStudent = null;
                currentlyRecognizedConf = 0;
                resetUnauthorizedState();
                document.getElementById('liveMatchAlert').style.display = 'none';
            }

            requestAnimationFrame(processAttendanceFrame);
        }

        async function submitManualAttendance() {
            if (!isAttCamRunning) {
                alert("⚠️ Please start the camera first!");
                return;
            }
            if (!isScannerActive) {
                alert("⚠️ Scanner is paused. Please click 'Resume Scanner'.");
                return;
            }

            const slotEval = evaluateCurrentSlot();
            if (!slotEval.allowed) {
                playBeep(300, 0.25);
                alert(`🚫 ATTENDANCE CLOSED!\n\nAttendance cannot be recorded outside allowed time slots.\n\nActive Window(s):\n${slotEval.ranges}`);
                return;
            }

            if (!currentlyRecognizedStudent || currentlyRecognizedConf < 95) {
                playBeep(300, 0.2);
                alert("❌ CANNOT MARK ATTENDANCE!\n\nNo recognized face meeting the required 95%+ confidence threshold detected.\nIf you are registered, please align clearly in good lighting.");
                return;
            }

            const student = currentlyRecognizedStudent;
            const confidence = currentlyRecognizedConf;
            const dateStr = getTodayDateString();
            const timeStr = new Date().toLocaleTimeString();

            // Post to Google Apps Script Backend
            const backendRes = await apiPost({
                action: 'markAttendance',
                studentId: student.id,
                roll: student.roll,
                name: student.name,
                dept: student.dept,
                date: dateStr,
                time: timeStr,
                slotType: slotEval.slotType,
                confidence: confidence + '%'
            });

            if (backendRes && backendRes.status === "already_marked") {
                alert(backendRes.message);
                return;
            }

            logAttendanceRecord(student, confidence, dateStr, timeStr, backendRes ? backendRes.message : "Success");

            playBeep(1200, 0.35);
            alert(`🎉 ATTENDANCE SUCCESSFUL!\n\nName: ${student.name}\nRoll No: ${student.roll}\nConfidence: ${confidence}%\nDate: ${dateStr}\nTime: ${timeStr}\n\n${backendRes ? backendRes.message : 'Saved to Google Sheets!'}`);

        function drawBox(ctx, box, label, color) {
            ctx.strokeStyle = color;
            ctx.lineWidth = 3;
            ctx.shadowColor = color;
            ctx.shadowBlur = 8;

            const cLen = 16;
            ctx.beginPath(); ctx.moveTo(box.x, box.y + cLen); ctx.lineTo(box.x, box.y); ctx.lineTo(box.x + cLen, box.y); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(box.x + box.w - cLen, box.y); ctx.lineTo(box.x + box.w, box.y); ctx.lineTo(box.x + box.w, box.y + cLen); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(box.x, box.y + box.h - cLen); ctx.lineTo(box.x, box.y + box.h); ctx.lineTo(box.x + cLen, box.y + box.h); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(box.x + box.w - cLen, box.y + box.h); ctx.lineTo(box.x + box.w, box.y + box.h); ctx.lineTo(box.x + box.w, box.y + box.h - cLen); ctx.stroke();

            ctx.fillStyle = color;
            ctx.fillRect(box.x, box.y - 26, ctx.measureText(label).width + 16, 22);
            ctx.shadowBlur = 0;
            ctx.fillStyle = "#000";
            ctx.font = "bold 11px sans-serif";
            ctx.fillText(label, box.x + 6, box.y - 10);
        }

        function logAttendanceRecord(student, confidence, dateStr, timeStr, statusMsg) {
            const existingIdx = attendanceLogs.findIndex(l => l.studentId === student.id && l.date === dateStr);
            
            const record = {
                id: 'LOG_' + Date.now(),
                studentId: student.id,
                roll: student.roll,
                name: student.name,
                dept: student.dept,
                date: dateStr || getTodayDateString(),
                time: timeStr || new Date().toLocaleTimeString(),
                timestamp: Date.now(),
                confidence: confidence + '%',
                statusMsg: statusMsg
            };

            if (existingIdx !== -1) {
                attendanceLogs[existingIdx] = record;
            } else {
                attendanceLogs.unshift(record);
            }

            localStorage.setItem('frs_v2_logs', JSON.stringify(attendanceLogs));
            renderTodayAttendance();
        }

                csv += `${idx + 1},"${l.date}","${l.time}","${l.roll}","${l.name}","${l.dept}","${l.confidence}","${statusStr}"\n`;
            });

            const encodedUri = encodeURI(csv);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", filename);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var action = "";
    var postData = {};

    if (e.parameter && e.parameter.action) {
      action = e.parameter.action;
    }

    if (e.postData && e.postData.contents) {
      try {
        postData = JSON.parse(e.postData.contents);
        if (postData.action) action = postData.action;
      } catch (err) {}
    }

    if (action === "getStudentsAndLogs") {
      return responseJSON(getStudentsAndLogs());
    } else if (action === "registerStudent") {
      return responseJSON(registerStudent(postData));
    } else if (action === "markAttendance") {
      return responseJSON(markAttendance(postData));
    } else if (action === "updateAdminPassword") {
      return responseJSON(updateAdminPassword(postData));
    } else {
      return responseJSON({ status: "error", message: "Invalid action request." });
    }
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// GET OR CREATE SPREADSHEET WITH 3 TABS: DIRECTORY, ATTENDANCE LOGS, SETTINGS
function getOrCreateSheets() {
  var year = new Date().getFullYear();
  var fileName = "Attendance_System_" + year;
  var files = DriveApp.getFilesByName(fileName);
  var ss;

  if (files.hasNext()) {
    ss = SpreadsheetApp.open(files.next());
  } else {
    ss = SpreadsheetApp.create(fileName);
  }

  // TAB 1: STUDENTS DIRECTORY
  var dirSheet = ss.getSheetByName("Students Directory");
  if (!dirSheet) {
    dirSheet = ss.insertSheet("Students Directory");
    
    var defaultSheet = ss.getSheetByName("Sheet1");
    if (defaultSheet && ss.getSheets().length > 1) {
      try { ss.deleteSheet(defaultSheet); } catch(e) {}
    }

    var dirHeaders = [
      "Student ID", 
      "Roll Number", 
      "Name", 
      "Department", 
      "Class / Section", 
      "Avatar Data", 
      "Biometric Descriptor"
    ];
    dirSheet.getRange(1, 1, 1, dirHeaders.length).setValues([dirHeaders]);
    dirSheet.getRange(1, 1, 1, dirHeaders.length)
      .setFontWeight("bold")
      .setBackground("#111827")
      .setFontColor("#06b6d4");
    dirSheet.setFrozenRows(1);
  }

  // TAB 2: ATTENDANCE LOGS
  var attSheet = ss.getSheetByName("Attendance Logs");
  if (!attSheet) {
    attSheet = ss.insertSheet("Attendance Logs");

    var attHeaders = ["Student ID", "Roll Number", "Name", "Department"];
    
    var startDate = new Date(year, 0, 1);
    var endDate = new Date(year, 11, 31);
    var curDate = new Date(startDate);

    while (curDate <= endDate) {
      var yyyy = curDate.getFullYear();
      var mm = String(curDate.getMonth() + 1).padStart(2, '0');
      var dd = String(curDate.getDate()).padStart(2, '0');
      attHeaders.push(yyyy + "-" + mm + "-" + dd);
      curDate.setDate(curDate.getDate() + 1);
    }

    attSheet.getRange(1, 1, 1, attHeaders.length).setValues([attHeaders]);
    attSheet.getRange(1, 1, 1, attHeaders.length)
      .setFontWeight("bold")
      .setBackground("#111827")
      .setFontColor("#06b6d4");
    attSheet.setFrozenRows(1);
    attSheet.setFrozenColumns(4);
  }

  // TAB 3: SETTINGS MANAGEMENT (Admin Password & App Config)
  var setSheet = ss.getSheetByName("Settings Management");
  if (!setSheet) {
    setSheet = ss.insertSheet("Settings Management");
    var setHeaders = ["Setting Key", "Setting Value"];
    setSheet.getRange(1, 1, 1, setHeaders.length).setValues([setHeaders]);
    setSheet.getRange(1, 1, 1, setHeaders.length)
      .setFontWeight("bold")
      .setBackground("#111827")
      .setFontColor("#06b6d4");
    setSheet.setFrozenRows(1);

    // Default Admin Password
    setSheet.appendRow(["Admin Password", "admin123"]);
  }

  return { dirSheet: dirSheet, attSheet: attSheet, setSheet: setSheet };
}

// REGISTER NEW STUDENT INTO "Students Directory"
function registerStudent(data) {
  var sheets = getOrCreateSheets();
  var dirSheet = sheets.dirSheet;
  var attSheet = sheets.attSheet;
  var lastRow = dirSheet.getLastRow();

  if (lastRow > 1) {
    var rolls = dirSheet.getRange(2, 2, lastRow - 1, 1).getValues();
    for (var i = 0; i < rolls.length; i++) {
      if (String(rolls[i][0]).toLowerCase() === String(data.roll).toLowerCase()) {
        return { status: "error", message: "Student with Roll Number (" + data.roll + ") is already registered!" };
      }
    }
  }

  var avatarStr = String(data.avatar || "");
  if (avatarStr.length > 45000) avatarStr = avatarStr.substring(0, 45000);

  var descriptorArr = [];
  if (Array.isArray(data.descriptor)) {
    descriptorArr = data.descriptor.map(function(val) {
      return typeof val === 'number' ? Math.round(val * 100) / 100 : 0;
    });
  }
  var descriptorStr = JSON.stringify(descriptorArr);
  if (descriptorStr.length > 45000) descriptorStr = descriptorStr.substring(0, 45000);

  var newRow = lastRow + 1;
  dirSheet.getRange(newRow, 1).setValue(String(data.id));
  dirSheet.getRange(newRow, 2).setValue(String(data.roll));
  dirSheet.getRange(newRow, 3).setValue(String(data.name));
  dirSheet.getRange(newRow, 4).setValue(String(data.dept));
  dirSheet.getRange(newRow, 5).setValue(String(data.classSec));
  
  var avatarRange = dirSheet.getRange(newRow, 6);
  avatarRange.setNumberFormat("@");
  avatarRange.setValue(avatarStr);

  var descRange = dirSheet.getRange(newRow, 7);
  descRange.setNumberFormat("@");
  descRange.setValue(descriptorStr);

  var attLastRow = attSheet.getLastRow();
  var newAttRow = attLastRow + 1;
  attSheet.getRange(newAttRow, 1).setValue(String(data.id));
  attSheet.getRange(newAttRow, 2).setValue(String(data.roll));
  attSheet.getRange(newAttRow, 3).setValue(String(data.name));
  attSheet.getRange(newAttRow, 4).setValue(String(data.dept));

  return { status: "success", message: "Registered successfully in Google Sheets!", student: data };
}

// RECORD ATTENDANCE - HANDLES ANYTIME (P), MORNING/EVENING HALF DAY (H), AND FULL DAY (P) COMBINATION
function markAttendance(data) {
  var sheets = getOrCreateSheets();
  var attSheet = sheets.attSheet;
  var lastRow = attSheet.getLastRow();

  if (lastRow < 2) {
    return { status: "error", message: "No registered students found in Google Sheet." };
  }

  var studentData = attSheet.getRange(2, 1, lastRow - 1, 2).getValues();
  var rowIndex = -1;

  for (var i = 0; i < studentData.length; i++) {
    if (String(studentData[i][0]) === String(data.studentId) || 
        String(studentData[i][1]).toLowerCase() === String(data.roll).toLowerCase()) {
      rowIndex = i + 2;
      break;
    }
  }

  if (rowIndex === -1) {
    rowIndex = lastRow + 1;
    attSheet.getRange(rowIndex, 1).setValue(String(data.studentId));
    attSheet.getRange(rowIndex, 2).setValue(String(data.roll));
    attSheet.getRange(rowIndex, 3).setValue(String(data.name));
    attSheet.getRange(rowIndex, 4).setValue(String(data.dept));
  }

  var headers = attSheet.getRange(1, 1, 1, attSheet.getLastColumn()).getValues()[0];
  var dateCol = -1;

  for (var j = 4; j < headers.length; j++) {
    var colHeaderStr = "";
    if (headers[j] instanceof Date) {
      var d = headers[j];
      colHeaderStr = d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, '0') + "-" + String(d.getDate()).padStart(2, '0');
    } else {
      colHeaderStr = String(headers[j]).trim();
    }

    if (colHeaderStr === data.date) {
      dateCol = j + 1;
      break;
    }
  }

  if (dateCol === -1) {
    dateCol = headers.length + 1;
    attSheet.getRange(1, dateCol).setValue(data.date)
      .setFontWeight("bold")
      .setBackground("#111827")
      .setFontColor("#06b6d4");
  }

  var currentVal = String(attSheet.getRange(rowIndex, dateCol).getValue() || "").trim();
  var slotType = data.slotType || "ANYTIME"; // "ANYTIME", "MORNING", "EVENING"
  var timeStr = data.time || "";

  if (slotType === "ANYTIME") {
    if (currentVal !== "") {
      return { 
        status: "already_marked", 
        message: "Attendance ALREADY marked today for " + data.name + "! (Recorded: " + currentVal + ")" 
      };
    }
    var logVal = "P (" + timeStr + ")";
    attSheet.getRange(rowIndex, dateCol).setValue(logVal);
    return { status: "success", message: "Full Day Attendance Recorded (P)!", time: timeStr, date: data.date };
  } 
  else if (slotType === "MORNING") {
    if (currentVal === "") {
      var logVal = "H (M: " + timeStr + ")";
      attSheet.getRange(rowIndex, dateCol).setValue(logVal);
      return { status: "success", message: "Morning Attendance Recorded as Half Day (H)!", time: timeStr, date: data.date };
    } else if (currentVal.indexOf("M:") !== -1) {
      return { status: "already_marked", message: "Morning slot ALREADY marked today for " + data.name + "! (" + currentVal + ")" };
    } else if (currentVal.indexOf("E:") !== -1) {
      var logVal = "P (M: " + timeStr + " | " + currentVal.replace(/^H\s*\(?/, "").replace(/\)$/, "") + ")";
      attSheet.getRange(rowIndex, dateCol).setValue(logVal);
      return { status: "success", message: "Full Day Attendance Completed (P)! (Morning + Evening recorded)", time: timeStr, date: data.date };
    } else {
      return { status: "already_marked", message: "Attendance ALREADY marked today for " + data.name + "! (" + currentVal + ")" };
    }
  } 
  else if (slotType === "EVENING") {
    if (currentVal === "") {
      var logVal = "H (E: " + timeStr + ")";
      attSheet.getRange(rowIndex, dateCol).setValue(logVal);
      return { status: "success", message: "Evening Attendance Recorded as Half Day (H)!", time: timeStr, date: data.date };
    } else if (currentVal.indexOf("E:") !== -1) {
      return { status: "already_marked", message: "Evening slot ALREADY marked today for " + data.name + "! (" + currentVal + ")" };
    } else if (currentVal.indexOf("M:") !== -1) {
      var logVal = "P (" + currentVal.replace(/^H\s*\(?/, "").replace(/\)$/, "") + " | E: " + timeStr + ")";
      attSheet.getRange(rowIndex, dateCol).setValue(logVal);
      return { status: "success", message: "Full Day Attendance Completed (P)! (Morning + Evening recorded)", time: timeStr, date: data.date };
    } else {
      return { status: "already_marked", message: "Attendance ALREADY marked today for " + data.name + "! (" + currentVal + ")" };
    }
  }
}

// UPDATE ADMIN PASSWORD IN SETTINGS TAB
function updateAdminPassword(data) {
  var sheets = getOrCreateSheets();
  var setSheet = sheets.setSheet;
  var lastRow = setSheet.getLastRow();

  if (!data.newPassword || String(data.newPassword).trim() === "") {
    return { status: "error", message: "New password cannot be empty!" };
  }

  var found = false;
  if (lastRow >= 2) {
    var keys = setSheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < keys.length; i++) {
      if (String(keys[i][0]).toLowerCase() === "admin password") {
        setSheet.getRange(i + 2, 2).setValue(String(data.newPassword).trim());
        found = true;
        break;
      }
    }
  }

  if (!found) {
    setSheet.appendRow(["Admin Password", String(data.newPassword).trim()]);
  }

  return { status: "success", message: "Admin Password updated successfully!" };
}

// FETCH STUDENTS, HISTORICAL LOGS & ADMIN PASSWORD FROM GOOGLE SHEET
function getStudentsAndLogs() {
  var sheets = getOrCreateSheets();
  var dirSheet = sheets.dirSheet;
  var attSheet = sheets.attSheet;
  var setSheet = sheets.setSheet;

  var dirLastRow = dirSheet.getLastRow();
  var attLastRow = attSheet.getLastRow();
  var attLastCol = attSheet.getLastColumn();
  var setLastRow = setSheet.getLastRow();

  var students = [];
  var logs = [];
  var adminPassword = "admin123";

  // Read Admin Password
  if (setLastRow >= 2) {
    var setVals = setSheet.getRange(2, 1, setLastRow - 1, 2).getValues();
    for (var m = 0; m < setVals.length; m++) {
      if (String(setVals[m][0]).toLowerCase() === "admin password") {
        adminPassword = String(setVals[m][1]).trim();
        break;
      }
    }
  }

  // Parse Students Directory
  if (dirLastRow >= 2) {
    var dirData = dirSheet.getRange(2, 1, dirLastRow - 1, 7).getValues();
    for (var i = 0; i < dirData.length; i++) {
      var row = dirData[i];
      var descriptor = [];

      try {
        descriptor = JSON.parse(row[6]);
      } catch(e) {}

      students.push({
        id: String(row[0]),
        roll: String(row[1]),
        name: String(row[2]),
        dept: String(row[3]),
        classSec: String(row[4]),
        avatar: String(row[5]),
        descriptor: descriptor
      });
    }
  }

  // Parse Attendance Logs
  if (attLastRow >= 2 && attLastCol >= 5) {
    var headers = attSheet.getRange(1, 1, 1, attLastCol).getValues()[0];
    var attData = attSheet.getRange(2, 1, attLastRow - 1, attLastCol).getValues();

    for (var k = 0; k < attData.length; k++) {
      var aRow = attData[k];
      var studentId = String(aRow[0]);
      var roll = String(aRow[1]);
      var name = String(aRow[2]);
      var dept = String(aRow[3]);

      for (var colIdx = 4; colIdx < headers.length; colIdx++) {
        var val = String(aRow[colIdx]).trim();
        if (val !== "") {
          var dateHeader = headers[colIdx];
          var dateStr = "";

          if (dateHeader instanceof Date) {
            dateStr = dateHeader.getFullYear() + "-" + String(dateHeader.getMonth() + 1).padStart(2, '0') + "-" + String(dateHeader.getDate()).padStart(2, '0');
          } else {
            dateStr = String(dateHeader).trim();
          }

          logs.push({
            id: "LOG_" + studentId + "_" + dateStr,
            studentId: studentId,
            roll: roll,
            name: name,
            dept: dept,
            date: dateStr,
            time: val,
            timestamp: new Date(dateStr).getTime() || Date.now(),
            confidence: "95%+"
          });
        }
      }
    }
  }

  return { status: "success", students: students, logs: logs, adminPassword: adminPassword };
}
