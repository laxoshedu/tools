        // 1. HARDCODED 5 DEFAULT ENGLISH QUESTIONS (INSTANT LOAD)
        const defaultEnglishQuestions = [
            {
                question: "Which word is a synonym of 'Happy'?",
                options: ["Joyful", "Sad", "Angry", "Tired"],
                correct: 0,
                explanation: "'Joyful' means feeling or expressing great happiness."
            },
            {
                question: "Choose the correctly spelled word:",
                options: ["Neccessary", "Necessary", "Necesary", "Neccesary"],
                correct: 1,
                explanation: "The correct spelling is 'Necessary' (one C, two S's)."
            },
            {
                question: "What is the past tense of the verb 'Go'?",
                options: ["Gone", "Going", "Went", "Goes"],
                correct: 2,
                explanation: "The simple past tense of 'go' is 'went'."
            },
            {
                question: "Identify the adjective in: 'She wore a beautiful dress.'",
                options: ["She", "Wore", "Beautiful", "Dress"],
                correct: 2,
                explanation: "'Beautiful' is an adjective modifying the noun 'dress'."
            },
            {
                question: "Complete the sentence: 'Neither of the boys ___ present.'",
                options: ["was", "were", "are", "have"],
                correct: 0,
                explanation: "'Neither' is singular and takes the singular verb 'was'."
            }
        ];

        // State Variables
        let currentSubject = "English";
        let quizQuestions = [...defaultEnglishQuestions];
        let currentQuestion = 0;
        let userAnswers = new Array(defaultEnglishQuestions.length).fill(null);
        let backendSubjects = [];

        // Auto-scroll Animation Engine State
        let isUserInteracting = false;

        // DOM Elements
        const chipsContainer = document.getElementById('chips-container');
        const chipsTrack = document.getElementById('chips-track');
        const quizView = document.getElementById('quiz-view');
        const resultScreen = document.getElementById('result-screen');
        const currentSubjectTitle = document.getElementById('current-subject-title');
        const tracker = document.getElementById('question-tracker');
        const progressBar = document.getElementById('progress-bar');
        const questionTitle = document.getElementById('question-title');
        const optionsGrid = document.getElementById('options-grid');
        const explanationCard = document.getElementById('explanation-card');
        const explanationText = document.getElementById('explanation-text');
        const prevBtn = document.getElementById('prev-btn');
        const nextBtn = document.getElementById('next-btn');
        const backendStatus = document.getElementById('backend-status');

        // Check Daily Completion State
        function isSubjectCompletedToday(subject) {
            const today = new Date().toISOString().split('T')[0];
            return localStorage.getItem(`quiz_completed_${subject}`) === today;
        }

        function setSubjectCompletedToday(subject) {
            const today = new Date().toISOString().split('T')[0];
            localStorage.setItem(`quiz_completed_${subject}`, today);
        }

        // Initialize App
        function initApp() {
            renderSubjectChips();
            loadQuestion();
            fetchBackendSubjects();
            setupInfiniteScrollEngine();
        }

        // Render Subject Chips (Duplicated for seamless continuous loop)
        function renderSubjectChips() {
            const baseSubjects = ["English", ...backendSubjects];
            chipsTrack.innerHTML = '';

            // Duplicate array to enable true continuous seamless looping
            const loopSubjects = [...baseSubjects, ...baseSubjects];

            loopSubjects.forEach((subject) => {
                const chip = document.createElement('div');
                const isCompleted = isSubjectCompletedToday(subject);
                const isActive = subject === currentSubject;

                chip.className = `subject-chip ${isActive ? 'active' : ''} ${isCompleted ? 'completed' : ''}`;
                chip.innerHTML = `${isCompleted ? '✔ ' : ''}${subject}`;
                chip.onclick = (e) => {
                    e.stopPropagation();
                    selectSubject(subject);
                };

                chipsTrack.appendChild(chip);
            });
        }

        // --- INFINITE CONTINUOUS SMOOTH SCROLL ENGINE ---
        function setupInfiniteScrollEngine() {
            const scrollSpeed = 0.5; // Slow, elegant speed

            function scrollStep() {
                if (!isUserInteracting && chipsContainer && chipsTrack.scrollWidth > 0) {
                    chipsContainer.scrollLeft += scrollSpeed;

                    // Halfway threshold: reset seamlessly back to start
                    const halfWidth = chipsTrack.scrollWidth / 2;
                    if (chipsContainer.scrollLeft >= halfWidth) {
                        chipsContainer.scrollLeft -= halfWidth;
                    }
                }
                requestAnimationFrame(scrollStep);
            }

            requestAnimationFrame(scrollStep);

            // Pause scrolling on interaction
            chipsContainer.addEventListener('mouseenter', () => isUserInteracting = true);
            chipsContainer.addEventListener('mouseleave', () => isUserInteracting = false);
            
            chipsContainer.addEventListener('touchstart', () => isUserInteracting = true, { passive: true });
            chipsContainer.addEventListener('touchend', () => {
                setTimeout(() => isUserInteracting = false, 2000);
            });

            chipsContainer.addEventListener('wheel', () => {
                isUserInteracting = true;
                clearTimeout(chipsContainer.scrollTimer);
                chipsContainer.scrollTimer = setTimeout(() => isUserInteracting = false, 2500);
            }, { passive: true });
        }

        // Fetch Backend Subjects from Google Apps Script
        async function fetchBackendSubjects() {
            try {
                const response = await fetch(`${APPS_SCRIPT_URL}?action=getSubjects`);
                const data = await response.json();

                if (data.success && data.subjects.length > 0) {
                    backendSubjects = data.subjects.filter(s => s !== "English");
                    backendStatus.innerText = "Synced";
                    renderSubjectChips();
                } else {
                    backendStatus.innerText = "Connected";
                }
            } catch (err) {
                console.log("Backend offline or URL not configured yet:", err);
                backendStatus.innerText = "Default Mode";
            }
        }

        // Handle Subject Selection
        async function selectSubject(subject) {
            if (currentSubject === subject && quizView.style.display !== 'none') return;

            if (isSubjectCompletedToday(subject)) {
                alert(`You have completed today's questions for "${subject}". Please choose another subject!`);
                return;
            }

            currentSubject = subject;
            renderSubjectChips();
            resultScreen.style.display = 'none';
            quizView.style.display = 'block';

            if (subject === "English") {
                quizQuestions = [...defaultEnglishQuestions];
                userAnswers = new Array(quizQuestions.length).fill(null);
                currentQuestion = 0;
                loadQuestion();
            } else {
                questionTitle.innerText = "LOADING...";
                optionsGrid.innerHTML = '<div class="spinner"></div>';
                explanationCard.classList.remove('show');

                try {
                    const response = await fetch(`${APPS_SCRIPT_URL}?action=getQuestions&subject=${encodeURIComponent(subject)}&count=10`);
                    const data = await response.json();

                    if (data.success && data.questions.length > 0) {
                        quizQuestions = data.questions;
                        userAnswers = new Array(quizQuestions.length).fill(null);
                        currentQuestion = 0;
                        loadQuestion();
                    } else {
                        questionTitle.innerText = `No questions found in tab "${subject}".`;
                        optionsGrid.innerHTML = '';
                    }
                } catch (err) {
                    console.error("Error fetching subject questions:", err);
                    questionTitle.innerText = "Failed to load questions from backend.";
                    optionsGrid.innerHTML = '';
                }
            }
        }

        // Load Active Question
        function loadQuestion() {
            const q = quizQuestions[currentQuestion];
            const savedAns = userAnswers[currentQuestion];

            currentSubjectTitle.innerText = `Subject: ${currentSubject}`;
            tracker.innerText = `Question ${currentQuestion + 1}/${quizQuestions.length}`;
            questionTitle.innerText = q.question;
            progressBar.style.width = `${((currentQuestion + 1) / quizQuestions.length) * 100}%`;

            optionsGrid.innerHTML = '';
            explanationCard.classList.remove('show');

            const prefixes = ['A', 'B', 'C', 'D'];

            q.options.forEach((opt, idx) => {
                const chip = document.createElement('div');
                chip.className = 'option-chip';
                chip.innerHTML = `
                    <div class="option-content">
                        <span class="option-prefix">${prefixes[idx]}</span>
                        <span>${opt}</span>
                    </div>
                    <span class="status-icon"></span>
                `;
                chip.onclick = () => selectOption(idx);
                optionsGrid.appendChild(chip);
            });

            if (savedAns !== null) {
                applyAnswerUI(savedAns);
            }

            updateNavButtons();
        }

        // Handle Instant Option Selection
        function selectOption(index) {
            if (userAnswers[currentQuestion] !== null) return;

            userAnswers[currentQuestion] = index;
            applyAnswerUI(index);
            updateNavButtons();
        }

        function applyAnswerUI(selectedIndex) {
            const q = quizQuestions[currentQuestion];
            const chips = optionsGrid.querySelectorAll('.option-chip');

            chips.forEach((chip, idx) => {
                chip.classList.add('disabled');

                if (idx === q.correct) {
                    chip.classList.add('correct');

// Entry point for web API calls
function doGet(e) {
  // Ensure database sheets and headers are ready
  autoSetupDatabase();

  var action = e ? e.parameter.action : null;
  var response = {};

  try {
    if (action === "getSubjects") {
      response = { success: true, subjects: getSubjectsList() };
    } else if (action === "getQuestions") {
      var subject = e.parameter.subject || "General Tech";
      var count = parseInt(e.parameter.count, 10) || 10;
      response = { 
        success: true, 
        subject: subject, 
        questions: getQuestionsForSubject(subject, count) 
      };
    } else {
      response = { 
        success: true, 
        message: "Quiz API active", 
        subjects: getSubjectsList() 
      };
    }
  } catch (err) {
    response = { success: false, error: err.toString() };
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

/**

 */
function getSubjectsList() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  var subjects = [];
  
  sheets.forEach(function(sheet) {
    subjects.push(sheet.getName());
  });

/**
 * Parses option letters (A, B, C, D) or numbers into 0-indexed integer.
 */
function parseCorrectOption(val) {
  if (val === undefined || val === null) return 0;
  var str = String(val).trim().toUpperCase();
  if (str === "A") return 0;
  if (str === "B") return 1;
  if (str === "C") return 2;
  if (str === "D") return 3;
  
  var num = parseInt(str, 10);
  if (!isNaN(num)) return num > 0 ? num - 1 : num;
  return 0;
}

/**
 * Fetches questions with continuous looping support (unless Disable Loop is checked).
 */
function getQuestionsForSubject(subject, count) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(subject);
  
  if (!sheet) {
    throw new Error("Subject tab '" + subject + "' not found.");
  }

  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return []; // Header only

  var rows = data.slice(1); // Omit header row
  var questions = [];

  // Check if Disable Loop setting is checked in Column H (Index 7) of row 2 or any row
  var disableLoop = false;
  if (rows.length > 0 && (rows[0][7] === true || String(rows[0][7]).toUpperCase() === "TRUE")) {
    disableLoop = true;
  }

  rows.forEach(function(row, idx) {
    if (row[0] && row[0].toString().trim() !== "") {
      questions.push({
        id: idx + 1,
        question: row[0],
        options: [row[1], row[2], row[3], row[4]],
        correct: parseCorrectOption(row[5]),
        explanation: row[6] || "No explanation provided."
      });
    }
  });

  if (questions.length === 0) return [];

  // Shuffle pool of questions
  for (var i = questions.length - 1; i > 0; i--) {
    var j = Math.floor(Math.random() * (i + 1));
    var temp = questions[i];
    questions[i] = questions[j];
    questions[j] = temp;
  }

  // Handle Looping logic:
  // If requested count > available questions and disableLoop is FALSE, repeat questions from beginning!
  var result = [];
  if (disableLoop) {
    // No repeat allowed -> return only up to available length
    result = questions.slice(0, count);
  } else {
    // Loop allowed -> fill up to `count` items by repeating
    var poolIndex = 0;
    while (result.length < count) {
      result.push(questions[poolIndex % questions.length]);
      poolIndex++;
    }
  }

  return result;
}
